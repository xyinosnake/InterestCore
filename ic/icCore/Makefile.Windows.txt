SHELL    = cmd.exe
CC       = clang.exe
OUT_DIR  = .\__BIN
CC_FLAGS = -std=c17
MACROS   = -DIC_USE_HIDE
TARGETS  = \
	$(OUT_DIR)\icCore.dll

.PHONY: all, clean
all: $(TARGETS)

$(OUT_DIR)\icCore.dll: .\icCore-Shared.c
	$(CC) .\icCore-Shared.c $(CC_FLAGS) -O3 -shared -o $(OUT_DIR)\icCore.dll

clean:
	del $(OUT_DIR)\*.*
